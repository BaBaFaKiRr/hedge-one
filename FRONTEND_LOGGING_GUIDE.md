# Frontend Logging Integration Guide

This guide explains how to fetch and display broker session logs from Supabase Storage in your frontend application.

## Architecture Overview

```
Frontend
  ↓ (queries metadata)
Supabase Postgres (broker_log_files table)
  ↓ (gets storage_path)
Supabase Storage (broker_logs bucket)
  ↓ (generates signed URL)
Frontend downloads & displays logs
```

**Key Principle:** Never query logs directly. Always:
1. Query metadata table for log file list
2. Generate signed URLs for each file
3. Download files from Storage
4. Parse and display NDJSON

---

## 1. Backend API Endpoints (Required)

You need to create these API endpoints in your backend:

### 1.1 List Log Files

**Endpoint:** `GET /api/logs/files`

**Query Parameters:**
- `user_id` (required) - User UUID
- `broker` (optional) - Filter by broker name
- `session_id` (optional) - Filter by session ID
- `log_date` (optional) - Filter by date (YYYY-MM-DD)
- `limit` (optional) - Max results (default: 100)

**Response:**
```json
{
  "files": [
    {
      "id": "uuid",
      "user_id": "uuid",
      "broker": "angelone",
      "session_id": "session_2025-01-14T09-12-03Z",
      "log_date": "2025-01-14",
      "log_hour": 9,
      "storage_path": "broker_logs/user_id/angelone/session_2025-01-14T09-12-03Z/2025-01-14/09-15.log",
      "created_at": "2025-01-14T09:15:00Z",
      "signed_url": "https://...signed-url..." // Generated by backend
    }
  ]
}
```

### 1.2 Get Log File Content

**Endpoint:** `GET /api/logs/content`

**Query Parameters:**
- `storage_path` (required) - Full storage path

**Response:**
```json
{
  "signed_url": "https://...",
  "expires_in": 3600
}
```

## 2. Backend Implementation (Python)

### 2.1 FastAPI Example

```python
from fastapi import FastAPI, Query
from helpers.log_retrieval import query_log_files, get_signed_url
from typing import Optional
from datetime import date

app = FastAPI()

@app.get("/api/logs/files")
async def list_log_files(
    user_id: str = Query(..., description="User UUID"),
    broker: Optional[str] = Query(None),
    session_id: Optional[str] = Query(None),
    log_date: Optional[str] = Query(None),  # YYYY-MM-DD format
    limit: int = Query(100, ge=1, le=1000)
):
    """
    List log files for a user with signed URLs.
    """
    # Parse date if provided
    date_obj = None
    if log_date:
        date_obj = date.fromisoformat(log_date)
    
    # Query metadata table
    files = query_log_files(
        user_id=user_id,
        broker=broker,
        session_id=session_id,
        log_date=date_obj,
        limit=limit
    )
    
    # Add signed URLs (expires in 1 hour)
    for file_info in files:
        storage_path = file_info.get("storage_path")
        if storage_path:
            signed_url = get_signed_url(storage_path, expires_in=3600)
            file_info["signed_url"] = signed_url
    
    return {"files": files}

@app.get("/api/logs/content")
async def get_log_content(
    storage_path: str = Query(..., description="Full storage path")
):
    """
    Get signed URL for log file.
    """
    signed_url = get_signed_url(storage_path, expires_in=3600)
    
    return {
        "signed_url": signed_url,
        "expires_in": 3600
    }
```

## 3. Frontend Implementation

### 3.1 TypeScript Types

```typescript
// types.ts
interface LogFile {
  id: string;
  user_id: string;
  broker: string;
  session_id: string;
  log_date: string;
  log_hour: number;
  storage_path: string;
  created_at: string;
  signed_url?: string;
}

interface LogEntry {
  ts: string;
  level: 'INFO' | 'WARNING' | 'ERROR' | 'DEBUG';
  component: string;
  message: string;
  strategy?: string;
  symbol?: string;
  order_id?: string;
}
```

### 3.2 React Hooks

```typescript
// hooks/useLogs.ts
import { useState, useEffect } from 'react';

export function useLogFiles(userId: string, broker?: string) {
  const [files, setFiles] = useState<LogFile[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    async function fetchFiles() {
      try {
        setLoading(true);
        const params = new URLSearchParams({
          user_id: userId,
          limit: '100'
        });
        if (broker) params.append('broker', broker);

        const response = await fetch(`/api/logs/files?${params}`);
        if (!response.ok) throw new Error('Failed to fetch log files');

        const data = await response.json();
        setFiles(data.files);
      } catch (err) {
        setError(err instanceof Error ? err.message : 'Unknown error');
      } finally {
        setLoading(false);
      }
    }

    if (userId) {
      fetchFiles();
    }
  }, [userId, broker]);

  return { files, loading, error };
}

export function useLogContent(signedUrl: string | null) {
  const [entries, setEntries] = useState<LogEntry[]>([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    async function fetchContent() {
      if (!signedUrl) {
        setEntries([]);
        return;
      }

      try {
        setLoading(true);
        const response = await fetch(signedUrl);
        if (!response.ok) throw new Error('Failed to fetch log content');

        const text = await response.text();
        
        // Parse NDJSON (newline-delimited JSON)
        const lines = text.trim().split('\n').filter(line => line.trim());
        const parsed = lines.map(line => JSON.parse(line) as LogEntry);
        
        setEntries(parsed);
      } catch (err) {
        setError(err instanceof Error ? err.message : 'Unknown error');
      } finally {
        setLoading(false);
      }
    }

    fetchContent();
  }, [signedUrl]);

  return { entries, loading, error };
}
```

          </div>

      </div>

      {/* Log Content Panel */}
      <div style={{ flex: 1, display: 'flex', flexDirection: 'column' }}>
        {selectedFile ? (
          <>
            {/* Header with Filters */}
            <div style={{ 
              padding: '1rem', 
              borderBottom: '1px solid #ddd',
              backgroundColor: '#f5f5f5'
            }}>
              <h3>
                {selectedFile.broker} - {selectedFile.log_date} {selectedFile.log_hour.toString().padStart(2, '0')}:00
              </h3>
              <div style={{ display: 'flex', gap: '1rem', marginTop: '0.5rem' }}>

## 5. Log Entry Structure

Each log entry is a JSON object in NDJSON format (one per line):

```json
{
  "ts": "2025-01-14T09:16:02.341Z",
  "level": "INFO",
  "component": "strategy.stock_ema_crossover",
  "message": "BUY order placed",
  "strategy": "stock_ema_crossover",
  "symbol": "NIFTY",
  "order_id": "A123456"
}
```

**Fields:**
- `ts`: ISO timestamp (UTC)
- `level`: Log level (INFO, WARNING, ERROR, DEBUG)
- `component`: Logger name (e.g., "strategy.stock_ema_crossover")
- `message`: Log message
- `strategy`: Strategy name (optional, extracted from component)
- `symbol`: Trading symbol (optional)
- `order_id`: Order ID (optional)

function VirtualizedLogViewer({ entries }: { entries: LogEntry[] }) {

## 7. Security Considerations

### 7.1 Authentication

- Always verify `user_id` matches authenticated user in backend
- Never expose service role key to frontend
- Use signed URLs with short expiration (1 hour)

## 8. Performance Tips

1. **Pagination**: Limit file list to 50-100 files
2. **Lazy Loading**: Only download log content when file is selected
3. **Virtual Scrolling**: Use for large log files (1000+ entries)
4. **Caching**: Cache signed URLs (they expire in 1 hour)
